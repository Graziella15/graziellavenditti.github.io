Application: Manipolazione del DOM

Scrivere un primo programma in JS che modifichi mediante JS il contenuto di un DIV o altro elemento. Linkare il programma come file .js esterno.

Questo è il testo originale del DIV. Usa i bottoni qui sotto per modificarmi!
👀 Clicca qui per ispezionare il codice JavaScript 
// Funzione 1: Modifica il contenuto HTML
function cambiaTesto() {
  const mioDiv = document.getElementById("target-div");
  mioDiv.innerHTML = "<strong>Magia! ✨</strong> Il contenuto è modificato.";
}

// Funzione 2: Modifica lo stile CSS
function cambiaStile() {
  const mioDiv = document.getElementById("target-div");
  mioDiv.style.backgroundColor = "#0284c7"; 
  mioDiv.style.color = "#ffffff"; 
  mioDiv.style.border = "3px dashed #38bdf8"; 
  mioDiv.style.borderRadius = "20px"; 
  mioDiv.style.transform = "scale(1.02)"; 
}

// Funzione 3: Nasconde o mostra
function nascondiMostra() {
  const mioDiv = document.getElementById("target-div");
  if (mioDiv.style.display === "none") {
    mioDiv.style.display = "block"; 
  } else {
    mioDiv.style.display = "none"; 
  }
}

// Funzione 4: Reset
function resetDiv() {
  const mioDiv = document.getElementById("target-div");
  mioDiv.innerHTML = "Questo è il testo originale...";
  mioDiv.style.backgroundColor = "";
  mioDiv.style.color = "";
  mioDiv.style.border = "";
  mioDiv.style.borderRadius = "";
  mioDiv.style.transform = "";
  mioDiv.style.display = "block";
}
